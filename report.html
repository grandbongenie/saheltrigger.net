Manifest Format

    "factsheet": { "url": "monthly-fact-sheet-2026-02.pdf", "title": "Monthly Fact Sheet - Feb 2026" }
  },
  "weekly": {
    "report": { "url": "weekly-report-2026-02-23.pdf", "title": "Weekly Report - 23 Feb 2026" }
  },
  "daily": {

  }
